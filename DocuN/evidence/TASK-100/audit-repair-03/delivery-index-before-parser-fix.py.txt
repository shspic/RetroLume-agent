def method_runs(batch,stage):
    path=EVIDENCE/batch/stage/'stderr.txt'
    runs=[]
    for number,line in enumerate(path.read_text('utf-8').splitlines(),1):
        match=re.fullmatch(r'(test_[A-Za-z0-9_]+) \(([^)]+)\) \.\.\. (ok|skipped .+|FAIL|ERROR)',line)
        if match:
            name=match.group(2)
            if not name.endswith('.'+match.group(1)):name+='.'+match.group(1)
            runs.append({'method':name,'verdict':match.group(3),'batch':batch,'raw':str(path),'line':number})
    return runs


def identities():
    inputs=load(ROOT/'DocuN/evidence/TASK-100/orchestration-receipt-04/continuation-input.json')
    checks=[]
    for item in inputs['objects']:
        path=Path(item['path']);actual=sha(path) if path.is_file() else None
        checks.append({**item,'post':actual,'unchanged':actual==item['actual']})
    for item in checks:
        if item['group']!='source_files' and not item['unchanged']:
            raise AssertionError('protected input drift: '+item['path'])
    pre={p.relative_to(EVIDENCE/'pre').as_posix():sha(p) for p in (EVIDENCE/'pre').rglob('*') if p.is_file()}
    current=hash_files(source_paths())
    dist=hash_files([p for p in (ROOT/'frontend/dist').rglob('*') if p.is_file()])
    full={**current,**dist}
    actual=load(EVIDENCE/B4/'as-run/manifest.json')
    assert full==actual, 'final bytes differ from final browser entry'
    backend={p:h for p,h in current.items() if p.startswith('backend/app/')}
    for batch in (B1,B3,B4):
        manifest=load(EVIDENCE/batch/'as-run-files.json')
        assert backend=={p:h for p,h in manifest.items() if p.startswith('backend/app/')}
    build=next(e['detail'] for e in reversed(Ledger().load()['events']) if e['event']=='frontend_build_pass')
    assert build['batch']==str(EVIDENCE/LAST_BUILD)
    assert build['dist']==dist
    for p,h in build['files'].items():
        if p.startswith('frontend/src/') or p=='frontend/package.json':assert current[p]==h
    assets=load(EVIDENCE/B4/'readonly-browser/served-assets.json')
    for asset in assets:
        relative='frontend/dist/'+asset['url'].split('18100/',1)[1]
        assert full[relative]==asset['sha256']
    post=EVIDENCE/'post';post.mkdir(exist_ok=False)
    for p,h in full.items():
        target=post/p;target.parent.mkdir(parents=True,exist_ok=True);shutil.copyfile(ROOT/p,target)
        assert sha(target)==h
    atomic_json(post/'manifest.json',full)
    changes=[]
    for p in sorted(set(pre)|set(full)):
        old,new=pre.get(p),full.get(p)
        if old==new:continue
        oldbytes=(EVIDENCE/'pre'/p).read_bytes() if old else b''
        newbytes=(ROOT/p).read_bytes() if new else b''
        destination=EVIDENCE/'diff'/(p+'.diff');destination.parent.mkdir(parents=True,exist_ok=True)
        try:
            difference=''.join(difflib.unified_diff(oldbytes.decode('utf-8').splitlines(True),
                newbytes.decode('utf-8').splitlines(True),fromfile='pre/'+p,tofile='post/'+p))
            destination.write_text(difference,encoding='utf-8',newline='')
        except UnicodeDecodeError:
            destination.write_text(json.dumps({'pre':old,'post':new,'binary_copies':True}),encoding='utf-8')
        changes.append({'path':p,'pre':old,'post':new,'kind':'modified' if old and new else 'added' if new else 'removed',
                        'diff':str(destination),'diff_sha256':sha(destination)})
    atomic_json(EVIDENCE/'final-identity.json',{'at':utc(),'baseline_input_objects':checks,'pre_files':pre,
        'post_files':full,'changes':changes,'backend_identical_as_run_batches':[B1,B3,B4],
        'final_source_and_dist_exact_as_run':B4,'final_build':LAST_BUILD,'consumed_bundle_assets':assets,
        'new_file_pre_absence':'new repair files have no invented predecessor bytes',
        'dist_removed_asset':'authorized Vite build replaced a generated bundle; pre bytes retained',
        'immutable_009_sha256':full['backend/migrations/009_teaching.sql']})
    return full,changes
