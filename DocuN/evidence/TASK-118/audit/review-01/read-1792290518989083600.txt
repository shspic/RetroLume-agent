6: async function sha256Hex(text: string): Promise<string> {
7:   const bytes = new TextEncoder().encode(text);
8:   const digest = await crypto.subtle.digest("SHA-256", bytes);
9:   return Array.from(new Uint8Array(digest)).map(b => b.toString(16).padStart(2, "0")).join("");
10: }

12: function canonical(value: unknown): string {
13:   const keys: string[] = [];
14:   const collect = (item: unknown) => {
15:     if (Array.isArray(item)) { item.forEach(collect); return; }
16:     if (typeof item === "object" && item !== null) {
17:       Object.keys(item as Record<string, unknown>).sort().forEach(k => keys.push(k));
18:       Object.values(item as Record<string, unknown>).forEach(collect);
19:     }
20:   };
21:   collect(value);
22:   // 直接用 JSON.stringify 的稳定实现：服务端 sort_keys + 紧凑分隔符。
23:   return stableStringify(value);
24: }

26: function stableStringify(value: unknown): string {
27:   if (value === null || typeof value !== "object") return JSON.stringify(value);
28:   if (Array.isArray(value)) return `[${value.map(stableStringify).join(",")}]`;
29:   const entries = Object.entries(value as Record<string, unknown>)
30:     .filter(([, v]) => v !== undefined)
31:     .sort(([a], [b]) => (a < b ? -1 : a > b ? 1 : 0));
32:   return `{${entries.map(([k, v]) => `${JSON.stringify(k)}:${stableStringify(v)}`).join(",")}}`;
33: }
