137: def decode_worker_request(value):
138:     exact(value, ("protocol", "call", "ledger", "context", "run_tag", "stop_ns", "call_seconds"))
139:     if type(value["protocol"]) is not int or value["protocol"] not in (3, 4):
140:         raise ValueError("worker protocol mismatch")
141:     call = OneCallInput.from_wire(value["call"])
142:     if call.protocol != value['protocol']: raise ValueError('capability_protocol_binding_invalid')
143:     context = CallContext.model_validate(value["context"], strict=True)
144:     validate_identity(context.activity_id, context.executor_id, call.operation_id, value["run_tag"])
145:     if call.request.input_result is not None and call.request.input_result.activity_id != context.activity_id:
146:         raise ValueError("worker input result activity mismatch")
147:     if context.operation_id != call.operation_id or context.execution_generation < 1:
148:         raise ValueError("worker identity mismatch")
149:     exact(value["ledger"], ("dsn", "connect_timeout_seconds", "statement_timeout_ms", "lock_timeout_ms") + (('data_guard',) if 'data_guard' in value['ledger'] else ()))
150:     dsn = value["ledger"]["dsn"]
151:     if type(dsn) is not str or not 1 <= len(dsn.encode("utf-8")) <= 4096 or "\0" in dsn:
152:         raise ValueError("invalid ledger configuration")
153:     seconds = value["call_seconds"]
154:     if type(seconds) not in (float, int) or not 0 < seconds <= 120 or call.service.timeout_seconds > seconds:
155:         raise ValueError("call timeout incompatible with session")
156:     if type(value["stop_ns"]) is not int or (value["stop_ns"] - time.monotonic_ns()) / 1e9 <= seconds + 0.25:
157:         raise ValueError("insufficient actual call window")
158:     return call, context
