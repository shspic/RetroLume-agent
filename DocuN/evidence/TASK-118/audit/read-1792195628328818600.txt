152:     """仅映射受信 bootstrap 和固定 app 的部署名字，逐个权限与授权者仍核对。"""

155:     for row in result['qualification_column_acls']:
156:         owner=owners.get(row['relname'])
157:         if owner is None:raise DataError('data_schema_not_ready',503)
158:         for field in ('column_acl','table_acl'):
159:             acl=row[field]
160:             if acl is None:continue
161:             entries=[]
162:             for entry in acl[1:-1].split(','):
163:                 match=re.fullmatch(r'([^=]+)=([^/]*)/(.+)',entry)
164:                 if not match or match[3]!=owner:raise DataError('data_schema_not_ready',503)
165:                 role,privileges=match[1],match[2]
166:                 if field=='table_acl':
167:                     if role==owner and privileges=='arwdDxtm':role='bootstrap_owner'
168:                     elif role==runtime_role and privileges=='arw':role='runtime_app'
169:                     else:raise DataError('data_schema_not_ready',503)
170:                 elif role!='retrolume_html_owner' or privileges!='r':raise DataError('data_schema_not_ready',503)
171:                 entries.append(role+'='+privileges+'/bootstrap_owner')
172:             row[field]='{'+','.join(entries)+'}'
